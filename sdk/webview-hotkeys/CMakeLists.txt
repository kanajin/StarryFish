cmake_minimum_required(VERSION 3.21)
project(GbfHotkeys LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
add_library(gbf_hotkeys SHARED native/core.cpp)
target_compile_definitions(gbf_hotkeys PRIVATE GBF_BUILD)
add_executable(sdk_checks checks/sdk.cpp)
target_link_libraries(sdk_checks PRIVATE gbf_hotkeys)
foreach(t gbf_hotkeys sdk_checks)
  target_compile_features(${t} PRIVATE cxx_std_17)
  target_compile_definitions(${t} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${t} PRIVATE /utf-8 /W4 /WX /EHsc)
  target_include_directories(${t} PRIVATE include vendor .build/generated)
  target_link_libraries(${t} PRIVATE user32 ole32 windowsapp bcrypt)
  set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endforeach()
